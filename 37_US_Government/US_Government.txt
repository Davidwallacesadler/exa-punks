|         XA          |
NOTE GET R/W COMMANDS
NOTE ALTERNATING R/W
NOTE THEN SEND R/W COMMS

GRAB 300
REPL GETCOMMANDS

MARK SENDNAMES
COPY F M
TEST EOF
FJMP SENDNAMES

COPY -1 M; DONE SIGNAL

DROP

REPL PERFORMCOMMANDS

MARK SENDCOMMANDLOOP
TEST X = 10
TJMP DONE

NOTE SEND READ
GRAB 400
SEEK X
COPY F M
DROP

NOTE SEND WRITE
GRAB 401
SEEK X
COPY F M
DROP

ADDI X 1 X; NEXT COMMAND
JUMP SENDCOMMANDLOOP

MARK DONE
COPY -1 M; DONE SIGNAL
HALT

NOTE -------------------



NOTE -- GET COMMANDS ---

MARK GETCOMMANDS
LINK 800
LINK 801
GRAB 200

MARK STARTGET
SEEK -9999
COPY M X; GET NAME
TEST X = -1; DONE SIGNAL
FJMP GETLOOP
HALT

MARK GETLOOP
TEST F = X
TJMP COPYCOMMANDS
JUMP GETLOOP

MARK COPYCOMMANDS
MODE; LOCAL COMMS
COPY 0 X
REPL COPYHELPER

MARK COPYLOOP
COPY F M
ADDI X 1 X
TEST X = 10
FJMP COPYLOOP
MODE; GLOBAL COMMS
JUMP STARTGET

MARK COPYHELPER
MAKE

MARK HELPERLOOP
COPY M F
ADDI X 1 X
TEST X = 10
FJMP HELPERLOOP

NOTE BRING FILE HOME
LINK -1
LINK -1
DROP
HALT

NOTE -------------------




NOTE -PERFORM COMMANDS-

MARK PERFORMCOMMANDS
MAKE
LINK 800

MARK COMMANDLOOP
NOTE GET COMMAND
COPY M F
SEEK -1
TEST F = -1
TJMP COMMANDDONE
SEEK -1

NOTE NAVIGATE
SWIZ F 3 T; DRIVE #
ADDI T 800 T
LINK T

NOTE CHECK R/W
MODI X 2 T
MODE; LOCAL COMMS
FJMP READ
JUMP WRITE

MARK READ
SEEK -1
SWIZ F 2 T; FILE #
ADDI T 200 T
REPL READHELPER

SEEK -1
SWIZ F 1 T; LINE #
MULI T 10 M

SEEK -1

MARK GETREADLOOP
COPY M F
SEEK -1
TEST F = -1
FJMP GETREADLOOP
SEEK -1
VOID F


MARK RESETCOMMANDLOOP
ADDI X 1 X
LINK -1
MODE; GLOBAL COMMS
JUMP COMMANDLOOP


MARK READHELPER
GRAB T
COPY M X; GET LINE #
SEEK X

MARK READHELPERLOOP
COPY F M
ADDI X 1 X
MODI X 10 T
TJMP READHELPERLOOP
COPY -1 M; DONE SIGNAL
JUMP COMMANDDONE

MARK WRITE
SEEK -1
COPY F T

REPL WRITEHELPER

SEEK -1
VOID F
SEEK -9999

MARK SENDWRITELOOP
COPY F M
TEST EOF
FJMP SENDWRITELOOP
SEEK -9999
JUMP RESETCOMMANDLOOP

MARK WRITEHELPER
SWIZ T 2 X
ADDI X 200 X
GRAB X

SWIZ T 1 X
MULI X 10 X

SEEK X

MARK WRITELOOP
COPY M F
ADDI X 1 X
MODI X 10 T
TJMP WRITELOOP

MARK COMMANDDONE
WIPE
